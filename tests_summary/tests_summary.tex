\medskip
\noindent\textbf{Модель.}
\[
  y_i = \beta_0 + \beta_1x_{1i} + \dots + \beta_{k-1}x_{k-1,i} + \varepsilon_i,
  \qquad i = 1, \dots, n,
\]
где $y_i$ — зависимая переменная, $x_{1i}, \dots, x_{k-1,i}$ — регрессоры,
$\varepsilon_i$ — ошибка, $n$ — число наблюдений, $k$ — число оцениваемых
параметров \emph{вместе} со свободным членом $\beta_0$. Суммы квадратов:
\[
  \TSS = \sum(y_i - \overline{y})^2,
  \qquad
  \RSS = \sum(\widehat y_i - \overline{y})^2,
  \qquad
  \ESS = \sum e_i^2,
  \qquad
  R^2 = \frac{\RSS}{\TSS} = 1 - \frac{\ESS}{\TSS},
\]
где $\RSS$ — объяснённая, $\ESS$ — остаточная сумма квадратов,
$e_i = y_i - \widehat y_i$ — остатки. Критические значения $t_\alpha$ и
$F_\alpha$ — в «Статистических таблицах» (таблицы 2 и 4).

% -------------------------------------------------------------------
\section{\texorpdfstring{$t$}{t}-тест: значимость коэффициента и сравнение с числом}
\[
  t = \frac{\widehat\beta_j - a}{s_{\widehat\beta_j}} \sim t(n - k),
  \qquad
  s_{\widehat\beta_j} = \sqrt{\widehat{\D}(\widehat\beta_j)},
\]
где $a$ — проверяемое значение коэффициента, $s_{\widehat\beta_j}$ —
стандартная ошибка оценки $\widehat\beta_j$. При $a = 0$ это проверка
значимости коэффициента.
\decision{$\beta_j = a$.}{$|t| > t_{\alpha/2}(n - k)$.}

% -------------------------------------------------------------------
\section{\texorpdfstring{$F$}{F}-тест: значимость регрессии в целом}
\[
  F = \frac{\RSS/(k - 1)}{\ESS/(n - k)}
    = \frac{R^2}{1 - R^2}\cdot\frac{n - k}{k - 1} \sim F(k - 1,\ n - k).
\]
\decision{$\beta_1 = \dots = \beta_{k-1} = 0$ (все коэффициенты, кроме
свободного члена).}{$F > F_\alpha(k - 1,\ n - k)$.}

% -------------------------------------------------------------------
\section{Тест на линейные ограничения}
Модель оценивают без ограничений (UR) и с ограничениями (R):
\[
  F = \frac{(\ESS_{\mathrm R} - \ESS_{\mathrm{UR}})/q}{\ESS_{\mathrm{UR}}/(n - k)}
    \sim F(q,\ n - k),
\]
где $\ESS_{\mathrm R}$, $\ESS_{\mathrm{UR}}$ — остаточные суммы квадратов
моделей с ограничениями и без, $q$ — число ограничений, $k$ — число
параметров модели \emph{без} ограничений. Если зависимая переменная в обеих
моделях одна и та же, то же самое можно записать через $R^2$:
\[
  F = \frac{(R^2_{\mathrm{UR}} - R^2_{\mathrm R})/q}{(1 - R^2_{\mathrm{UR}})/(n - k)} .
\]
\decision{ограничения верны.}{$F > F_\alpha(q,\ n - k)$.}

% -------------------------------------------------------------------
\section{Тест Чоу: одинакова ли зависимость в двух группах}
Регрессию оценивают по всей выборке и отдельно по каждой группе:
\[
  F = \frac{\bigl(\ESS_{\mathrm R} - (\ESS_1 + \ESS_2)\bigr)/k}
           {(\ESS_1 + \ESS_2)/(n_1 + n_2 - 2k)}
    \sim F(k,\ n_1 + n_2 - 2k),
\]
где $\ESS_{\mathrm R}$ — остаточная сумма квадратов регрессии по всей
выборке, $\ESS_1$, $\ESS_2$ — по первой и второй группам, $n_1$, $n_2$ —
число наблюдений в группах. Это частный случай теста на ограничения:
$q = k$, модель без ограничений — две отдельные регрессии, у неё $2k$
параметров.
\decision{все $k$ коэффициентов в группах одинаковы.}{$F > F_\alpha(k,\ n_1 + n_2 - 2k)$.}

% -------------------------------------------------------------------
\section{VIF: фактор инфляции дисперсии}
\[
  \mathrm{VIF}_j = \frac{1}{1 - R_j^2},
\]
где $R_j^2$ — коэффициент детерминации вспомогательной регрессии $x_j$ на
остальные регрессоры с константой. $\mathrm{VIF}_j$ показывает, во сколько
раз дисперсия $\widehat\beta_j$ больше, чем была бы при регрессоре, не
связанном с остальными. Правило: $\mathrm{VIF}_j > 10$ (то есть
$R_j^2 > 0{,}9$) — серьёзная мультиколлинеарность.

% -------------------------------------------------------------------
\section{\texorpdfstring{$F$}{F}-тест на мультиколлинеарность}
Это $F$-тест значимости той же вспомогательной регрессии: в ней $k - 2$
регрессора (все, кроме $x_j$) и константа, то есть $k - 1$ параметр:
\[
  F_j = \frac{R_j^2/(k - 2)}{(1 - R_j^2)/(n - k + 1)}
      = \frac{R_j^2}{1 - R_j^2}\cdot\frac{n - k + 1}{k - 2}
      \sim F(k - 2,\ n - k + 1).
\]
\decision{$x_j$ не связан линейно с остальными регрессорами.}{$F_j > F_\alpha(k - 2,\ n - k + 1)$.}
